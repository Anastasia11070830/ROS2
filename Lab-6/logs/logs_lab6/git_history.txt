+import rclpy
+from rclpy.node import Node
+from std_msgs.msg import String
+
+
+class Lab1Publisher(Node):
+
+    def __init__(self):
+        super().__init__('lab1_publisher')
+        self.publisher_ = self.create_publisher(String, 'lab1_topic', 10)
+        # Таймер з періодом 1 с — публікація 1 раз на секунду
+        self.timer = self.create_timer(1.0, self.timer_callback)
+        self.count = 0
+
+    def timer_callback(self):
+        msg = String()
+        msg.data = f'Hello from Lab1! Message #{self.count}'
+        self.publisher_.publish(msg)
+        self.get_logger().info(f'Publishing: "{msg.data}"')
+        self.count += 1
+
+
+def main(args=None):
+    rclpy.init(args=args)
+    node = Lab1Publisher()
+    try:
+        rclpy.spin(node)
+    except KeyboardInterrupt:
+        pass
+    finally:
+        node.destroy_node()
+        if rclpy.ok():
+            rclpy.shutdown()
+
+
+if __name__ == '__main__':
+    main()

+import rclpy
+from rclpy.node import Node
+from std_msgs.msg import Float32, String
+
+
+class MonitorNode(Node):
+
+    def __init__(self):
+        super().__init__('monitor_node')
+        self.declare_parameter('heartbeat_timeout', 2.5)  # с без heartbeat = нода не відповідає
+        self.declare_parameter('data_timeout', 2.5)       # с без даних = дані не приходять
+        self.declare_parameter('expected_rate', 10.0)     # Гц, очікувана частота /sensor_data
+        self.heartbeat_timeout = self.get_parameter('heartbeat_timeout').value
+        self.data_timeout = self.get_parameter('data_timeout').value
+        self.expected_rate = self.get_parameter('expected_rate').value
+
+        now = self.now()
+        self.last_heartbeat = now
+        self.last_data = now
+        self.last_check = now
+        self.data_count = 0
+        self.state = None
+
+        self.create_subscription(String, '/heartbeat', self.heartbeat_callback, 10)
+        self.create_subscription(Float32, '/sensor_data', self.data_callback, 10)
+        self.status_pub = self.create_publisher(String, '/system_status', 10)
+        # Раз на секунду оцінюємо стан системи
+        self.timer = self.create_timer(1.0, self.check)
+
+        self.get_logger().info(
+            f'Monitor Node started: heartbeat_timeout={self.heartbeat_timeout} s, '
+            f'data_timeout={self.data_timeout} s, expected_rate={self.expected_rate} Hz')
+
+    def now(self):
+        return self.get_clock().now().nanoseconds / 1e9
+
+    def heartbeat_callback(self, msg):
+        self.last_heartbeat = self.now()
+        self.get_logger().debug(f'Heartbeat received: {msg.data}')
+
+    def data_callback(self, msg):
+        self.last_data = self.now()
+        self.data_count += 1
+
+    def check(self):
+        now = self.now()
+        heartbeat_age = now - self.last_heartbeat
+        data_age = now - self.last_data
+        rate = self.data_count / (now - self.last_check)
+        self.data_count = 0
+        self.last_check = now
+
+        if heartbeat_age > self.heartbeat_timeout:
+            level = 'ERROR'
+            text = f'Heartbeat lost: no signal for {heartbeat_age:.1f} s, node is not responding'
+        elif data_age > self.data_timeout:
+            level = 'ERROR'
+            text = f'No data on /sensor_data for {data_age:.1f} s, but heartbeat is alive'
+        elif rate < 0.75 * self.expected_rate:
+            level = 'WARN'
+            text = f'Data unstable: rate {rate:.1f} Hz (expected {self.expected_rate:.1f} Hz)'
+        else:
+            level = 'OK'
+            text = f'System stable: heartbeat ok, data rate {rate:.1f} Hz'
+
+        if level == 'ERROR':
+            self.get_logger().error(text)
+        elif level == 'WARN':
+            self.get_logger().warn(text)
+        else:
+            self.get_logger().info(text)
+        if level != self.state:
+            if self.state is not None:
+                self.get_logger().info(f'Status changed: {self.state} -> {level}')
+            self.state = level
+
+        status = String()
+        status.data = f'{level}: {text}'
+        self.status_pub.publish(status)
+
+
+def main(args=None):
+    rclpy.init(args=args)
+    node = MonitorNode()
+    try:
+        rclpy.spin(node)
+    except KeyboardInterrupt:
+        pass
+    finally:
+        node.destroy_node()
+        if rclpy.ok():
+            rclpy.shutdown()
+
+
+if __name__ == '__main__':
+    main()

+import math
+
+import rclpy
+from rclpy.node import Node
+from geometry_msgs.msg import Twist
+from std_msgs.msg import Bool, String
+
+
+class MotionNode(Node):
+
+    def __init__(self):
+        super().__init__('motion_node')
+        # Межі швидкості задані параметрами — їх можна змінити без зміни коду
+        self.declare_parameter('max_linear_speed', 1.0)
+        self.declare_parameter('max_angular_speed', 1.5)
+        self.declare_parameter('output_topic', '/turtle1/cmd_vel')
+        self.max_linear = self.get_parameter('max_linear_speed').value
+        self.max_angular = self.get_parameter('max_angular_speed').value
+        output_topic = self.get_parameter('output_topic').value
+
+        self.sensor_alarm = False
+        self.subscription = self.create_subscription(
+            Twist, '/cmd_vel', self.cmd_vel_callback, 10)
+        self.alarm_subscription = self.create_subscription(
+            Bool, '/sensor_alarm', self.alarm_callback, 10)
+        self.publisher_ = self.create_publisher(Twist, output_topic, 10)
+        self.status_publisher = self.create_publisher(String, '/motion_status', 10)
+
+        self.get_logger().info(
+            f'Motion Node started: max_linear={self.max_linear} m/s, '
+            f'max_angular={self.max_angular} rad/s, output={output_topic}')
+
+    def limit(self, name, value, max_value):
+        if abs(value) <= max_value:
+            return value
+        # Обрізаємо до порогу, зберігаючи напрямок руху
+        limited = math.copysign(max_value, value)
+        self.get_logger().warn(
+            f'{name}={value:.2f} exceeds limit +/-{max_value:.2f}, '
+            f'limited to {limited:.2f}')
+        return limited
+
+    def alarm_callback(self, msg):
+        if msg.data != self.sensor_alarm:
+            self.sensor_alarm = msg.data
+            if self.sensor_alarm:
+                self.get_logger().warn('Sensor alarm ON: forward motion is blocked')
+            else:
+                self.get_logger().info('Sensor alarm OFF: motion allowed')
+
+    def cmd_vel_callback(self, msg):
+        self.get_logger().info(
+            f'Received: linear.x={msg.linear.x:.2f}, angular.z={msg.angular.z:.2f}')
+
+        cmd = Twist()
+        cmd.linear.x = self.limit('linear.x', msg.linear.x, self.max_linear)
+        cmd.angular.z = self.limit('angular.z', msg.angular.z, self.max_angular)
+        limited = (cmd.linear.x, cmd.angular.z) != (msg.linear.x, msg.angular.z)
+
+        # Тривога сенсора забороняє лише рух вперед; назад і поворот дозволені
+        blocked = self.sensor_alarm and cmd.linear.x > 0.0
+        if blocked:
+            cmd.linear.x = 0.0
+            self.get_logger().warn(
+                'Forward motion blocked by sensor alarm', throttle_duration_sec=1.0)
+
+        self.publisher_.publish(cmd)
+
+        self.get_logger().info(
+            f'Sent: linear.x={cmd.linear.x:.2f}, angular.z={cmd.angular.z:.2f}')
+
+        # Статус для GUI: OK, LIMITED (швидкість обмежено) або BLOCKED (рух заблоковано)
+        label = 'BLOCKED' if blocked else ('LIMITED' if limited else 'OK')
+        status = String()
+        status.data = (f'{label}: '
+                       f'linear.x={cmd.linear.x:.2f}, angular.z={cmd.angular.z:.2f}')
+        self.status_publisher.publish(status)
+
+
+def main(args=None):
+    rclpy.init(args=args)
+    node = MotionNode()
+    try:
+        rclpy.spin(node)
+    except KeyboardInterrupt:
+        pass
+    finally:
+        node.destroy_node()
+        if rclpy.ok():
+            rclpy.shutdown()
+
+
+if __name__ == '__main__':
+    main()

+import csv
+import os
+
+import rclpy
+from rclpy.node import Node
+from geometry_msgs.msg import Twist
+
+
+class PlaybackNode(Node):
+
+    def __init__(self):
+        super().__init__('playback_node')
+        self.declare_parameter('file', '~/ros2_ws/data/route.csv')
+        self.declare_parameter('speed', 1.0)        # 2.0 = відтворення вдвічі швидше
+        self.declare_parameter('start_delay', 3.0)  # с до початку відтворення
+        self.declare_parameter('end_hold', 1.0)     # с утримання останньої команди
+        path = os.path.expanduser(self.get_parameter('file').value)
+        self.speed = self.get_parameter('speed').value
+        self.end_hold = self.get_parameter('end_hold').value
+        start_delay = self.get_parameter('start_delay').value
+
+        try:
+            self.rows = self.load(path)
+        except (OSError, KeyError, ValueError) as err:
+            self.get_logger().error(f'Cannot read scenario {path}: {err}')
+            raise SystemExit(1)
+        if not self.rows:
+            self.get_logger().error(f'Scenario {path} is empty')
+            raise SystemExit(1)
+
+        self.publisher_ = self.create_publisher(Twist, '/cmd_vel', 10)
+        self.index = 0
+        self.errors = []
+        self.done = False
+        self.start = self.now() + start_delay
+        # Таймер 100 Гц перевіряє, чи настав час наступної команди
+        self.timer = self.create_timer(0.01, self.tick)
+
+        duration = self.rows[-1][0] / self.speed
+        self.get_logger().info(
+            f'Loaded {len(self.rows)} commands from {path}, duration {duration:.2f} s, '
+            f'speed x{self.speed}. Starting in {start_delay} s')
+
+    def now(self):
+        return self.get_clock().now().nanoseconds / 1e9
+
+    def load(self, path):
+        with open(path, newline='', encoding='utf-8') as f:
+            return [(float(r['time']), float(r['linear_x']), float(r['angular_z']))
+                    for r in csv.DictReader(f, skipinitialspace=True)]
+
+    def send(self, linear_x, angular_z):
+        msg = Twist()
+        msg.linear.x = linear_x
+        msg.angular.z = angular_z
+        self.publisher_.publish(msg)
+
+    def tick(self):
+        elapsed = self.now() - self.start
+        # Відправляємо всі команди, час яких уже настав, у порядку запису
+        while self.index < len(self.rows):
+            t, linear_x, angular_z = self.rows[self.index]
+            planned = t / self.speed
+            if elapsed < planned:
+                return
+            self.send(linear_x, angular_z)
+            self.errors.append(elapsed - planned)
+            self.index += 1
+            self.get_logger().info(
+                f'#{self.index}/{len(self.rows)} planned={planned:.3f} s, '
+                f'actual={elapsed:.3f} s, linear.x={linear_x:.3f}, angular.z={angular_z:.3f}')
+
+        # Усі команди відправлено: після утримання останньої надсилаємо Стоп
+        if elapsed >= self.rows[-1][0] / self.speed + self.end_hold:
+            self.send(0.0, 0.0)
+            self.timer.cancel()
+            self.done = True
+            worst = max(abs(e) for e in self.errors) * 1000.0
+            mean = sum(abs(e) for e in self.errors) / len(self.errors) * 1000.0
+            self.get_logger().info(
+                f'Playback finished: {len(self.errors)} commands sent, '
+                f'timing error mean={mean:.1f} ms, max={worst:.1f} ms. Stop command sent')
+
+
+def main(args=None):
+    rclpy.init(args=args)
+    node = PlaybackNode()
+    try:
+        while rclpy.ok() and not node.done:
+            rclpy.spin_once(node, timeout_sec=0.05)
+    except KeyboardInterrupt:
+        pass
+    finally:
+        node.destroy_node()
+        if rclpy.ok():
+            rclpy.shutdown()
+
+
+if __name__ == '__main__':
+    main()

+import os
+
+import rclpy
+from rclpy.node import Node
+from geometry_msgs.msg import Twist
+
+
+class RecordNode(Node):
+
+    def __init__(self):
+        super().__init__('record_node')
+        self.declare_parameter('file', '~/ros2_ws/data/route.csv')
+        path = os.path.expanduser(self.get_parameter('file').value)
+        os.makedirs(os.path.dirname(path), exist_ok=True)
+
+        # Файл відкривається на запис: попередній сценарій перезаписується
+        self.file = open(path, 'w', encoding='utf-8')
+        self.file.write('time, linear_x, angular_z\n')
+        self.start = None
+        self.count = 0
+
+        self.subscription = self.create_subscription(
+            Twist, '/cmd_vel', self.cmd_vel_callback, 10)
+        self.get_logger().info(f'Recording /cmd_vel to {path}. Press Ctrl+C to finish.')
+
+    def cmd_vel_callback(self, msg):
+        now = self.get_clock().now().nanoseconds / 1e9
+        if self.start is None:
+            # Час відраховується від першої отриманої команди
+            self.start = now
+        elapsed = now - self.start
+
+        self.file.write(f'{elapsed:.3f}, {msg.linear.x:.3f}, {msg.angular.z:.3f}\n')
+        # flush: дані потрапляють у файл одразу, навіть якщо ноду буде перервано
+        self.file.flush()
+        self.count += 1
+        self.get_logger().info(
+            f'#{self.count} t={elapsed:.3f} s, '
+            f'linear.x={msg.linear.x:.3f}, angular.z={msg.angular.z:.3f}')
+
+    def close(self):
+        self.file.close()
+        self.get_logger().info(f'Recording finished: {self.count} commands saved')
+
+
+def main(args=None):
+    rclpy.init(args=args)
+    node = RecordNode()
+    try:
+        rclpy.spin(node)
+    except KeyboardInterrupt:
+        pass
+    finally:
+        node.close()
+        node.destroy_node()
+        if rclpy.ok():
+            rclpy.shutdown()
+
+
+if __name__ == '__main__':
+    main()

+        self.declare_parameter('period', 24.0)              # с, період зміни відстані
+        self.declare_parameter('fault_probability', 0.0)    # імітація збоїв читання
+
+        self.rate = self.get_parameter('publish_rate').value
+        self.window = deque(maxlen=self.get_parameter('window_size').value)
+        self.noise_std = self.get_parameter('noise_std').value
+        self.spike_probability = self.get_parameter('spike_probability').value
+        self.period = self.get_parameter('period').value
+
+        self.raw_pub = self.create_publisher(Float32, '/sensor_data', 10)
+        self.filtered_pub = self.create_publisher(Float32, '/sensor_data_filtered', 10)
+        self.alarm_pub = self.create_publisher(Bool, '/sensor_alarm', 10)
+        self.heartbeat_pub = self.create_publisher(String, '/heartbeat', 10)
+
+        self.step = 0
+        self.alarm = False
+        self.heartbeat_count = 0
+        # Таймер задає періодичність публікації (10 Гц = кожні 0.1 с)
+        self.timer = self.create_timer(1.0 / self.rate, self.timer_callback)
+        # Heartbeat: сигнал активності раз на секунду
+        self.heartbeat_timer = self.create_timer(1.0, self.heartbeat_callback)
+
+        self.get_logger().info(
+            f'Sensor Node started: rate={self.rate} Hz, '
+            f'window={self.window.maxlen}, threshold={self.get_parameter("threshold").value} m')
+
+    def heartbeat_callback(self):
+        msg = String()
+        msg.data = f'sensor_node is alive, beat={self.heartbeat_count}'
+        self.heartbeat_pub.publish(msg)
+        self.get_logger().debug(f'Heartbeat sent: {msg.data}')
+        self.heartbeat_count += 1
+
+    def read_sensor(self):
+        # Імітація дальноміра: плавна зміна відстані + шум + рідкісні викиди
+        t = self.step / self.rate
+        value = 1.4 + 1.0 * math.sin(2.0 * math.pi * t / self.period)
+        value += random.gauss(0.0, self.noise_std)
+        if random.random() < self.spike_probability:
+            value = random.uniform(0.08, 0.2)
+        return max(value, 0.02)
+
+    def timer_callback(self):
+        # Імітація збою читання: дані не публікуються, у лог іде ERROR
+        if random.random() < self.get_parameter('fault_probability').value:
+            self.step += 1
+            self.get_logger().error(
+                'Sensor read failed: no response from device', throttle_duration_sec=1.0)
+            return
+
+        raw = self.read_sensor()
+        self.step += 1
+        self.get_logger().debug(f'Sample #{self.step}: raw={raw:.3f} m')
+
+        # Усереднення: середнє арифметичне останніх N вимірювань
+        self.window.append(raw)
+        filtered = sum(self.window) / len(self.window)
+
+        self.raw_pub.publish(Float32(data=raw))
+        self.filtered_pub.publish(Float32(data=filtered))
+
+        # Поріг читається щоразу, тож його можна змінювати командою ros2 param set
+        threshold = self.get_parameter('threshold').value
+        critical = filtered < threshold
+        if critical:
+            self.get_logger().warn(
+                f'CRITICAL: filtered={filtered:.2f} m is below threshold {threshold:.2f} m',
+                throttle_duration_sec=1.0)
+        elif self.alarm:
+            self.get_logger().info(f'Value back to normal: filtered={filtered:.2f} m')
+        self.alarm = critical
+        self.alarm_pub.publish(Bool(data=self.alarm))
+
+        self.get_logger().info(
+            f'raw={raw:.2f} m, filtered={filtered:.2f} m', throttle_duration_sec=1.0)
+
+
+def main(args=None):
+    rclpy.init(args=args)
+    node = SensorNode()
+    try:
+        rclpy.spin(node)
+    except KeyboardInterrupt:
+        pass
+    finally:
+        node.get_logger().info('Sensor Node stopped')
+        node.destroy_node()
+        if rclpy.ok():
+            rclpy.shutdown()
+
+
+if __name__ == '__main__':
+    main()
